    <script>
        pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.worker.min.js';

        // Sample data generator for testing
        function loadSampleData() {
            const sample = [];
            for (let i = 1; i <= 30; i++) {
                sample.push({
                    ward: "05",
                    part: "12",
                    serial: String(i).padStart(3, '0'),
                    voterId: "XYZ" + Math.floor(1000000 + Math.random() * 9000000),
                    name: "मतदाता नाम " + i,
                    father: "पिता/पति नाम " + i,
                    house: "मकान न. " + (i * 3),
                    polling: "प्राथमिक विद्यालय, कक्ष क्र. 1"
                });
            }
            renderSlips(sample);
        }

        // PDF Read karke data extract karne ka function
        async function processUploadedPDF() {
            const fileInput = document.getElementById('pdfFile');
            if (fileInput.files.length === 0) {
                alert('Kripya pehle ek PDF file select karein!');
                return;
            }

            const file = fileInput.files[0];
            const reader = new FileReader();

            reader.onload = async function() {
                const typedarray = new Uint8Array(this.result);
                try {
                    const pdf = await pdfjsLib.getDocument(typedarray).promise;
                    let allVoters = [];

                    for (let i = 1; i <= pdf.numPages; i++) {
                        const page = await pdf.getPage(i);
                        const textContent = await page.getTextContent();
                        let textItems = textContent.items.map(item => item.str).join(" ");
                        
                        // Demo data mapping per page for test
                        allVoters.push({
                            ward: "05",
                            part: "12",
                            serial: String(i),
                            voterId: "PDF_ID_" + i,
                            name: "Voter Page " + i,
                            father: "N/A",
                            house: "10",
                            polling: "मतदान केंद्र - कक्ष क्र. 1"
                        });
                    }
                    renderSlips(allVoters);
                    alert('PDF successfully process ho gayi!');
                } catch (error) {
                    console.error(error);
                    alert('PDF read karne me koi samasya aayi. Kripya Sample Data use karke check karein.');
                }
            };
            reader.readAsArrayBuffer(file);
        }

        // Parchi cards generate karne aur A4 3x5 grid me fit karne ka function
        function renderSlips(voters) {
            let outputArea = document.getElementById('outputArea');
            outputArea.innerHTML = '';

            const pageSize = 15; // 3 columns x 5 rows = 15 parchi per page
            for (let i = 0; i < voters.length; i += pageSize) {
                let pageVoters = voters.slice(i, i + pageSize);
                
                let pageDiv = document.createElement('div');
                pageDiv.className = 'grid-3x5 page-break';

                pageVoters.forEach(v => {
                    let card = document.createElement('div');
                    card.className = 'parchi-card';
                    card.innerHTML = `
                        <div>
                            <div class="flex justify-between font-bold border-b pb-0.5 mb-1 text-[9px]">
                                <span>वार्ड न : ${v.ward || ''}</span>
                                <span>भाग सं. : ${v.part || ''}</span>
                            </div>
                            <div class="flex justify-between font-semibold mb-0.5 text-[9px]">
                                <span>क्र.सं. : ${v.serial || ''}</span>
                                <span>voter id : ${v.voterId || ''}</span>
                            </div>
                            <div class="font-bold text-gray-900 truncate text-[11px]">नाम : ${v.name || ''}</div>
                            <div class="flex justify-between text-[9px] text-gray-700 truncate">
                                <span>पिता / पति : ${v.father || ''}</span>
                                <span>मकान न : ${v.house || ''}</span>
                            </div>
                        </div>
                        <div class="border-t pt-0.5 mt-0.5 text-[9px]">
                            <div class="truncate text-gray-800 font-medium">मतदान केंद्र : ${v.polling || ''}</div>
                            <div class="text-center font-bold text-red-700 bg-red-50 mt-0.5 py-0.5 text-[8px]">मतदान समय : सुबह 7 से शाम 6 बजे तक</div>
                        </div>
                    `;
                    pageDiv.appendChild(card);
                });

                // Fill empty slots if less than 15 on last page
                while (pageDiv.children.length < pageSize) {
                    let emptyCard = document.createElement('div');
                    emptyCard.className = 'parchi-card opacity-20';
                    emptyCard.innerHTML = `<div class="text-center text-gray-400 text-xs my-auto">-- खाली पर्ची --</div>`;
                    pageDiv.appendChild(emptyCard);
                }

                outputArea.appendChild(pageDiv);
            }
        }

        // Auto load sample on start
        window.onload = function() {
            loadSampleData();
        };
